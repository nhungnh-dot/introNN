  function initTarget(x1,x2){
    return sig(initTeacher.w1*x1+initTeacher.w2*x2);

  function initLossGrad(w1,w2,data){
    let loss=0,g1=0,g2=0;
    data.forEach(([x1,x2])=>{
      const t=initTarget(x1,x2), y=sig(w1*x1+w2*x2), e=y-t;
      loss+=e*e;
      const d=2*e*y*(1-y);
      g1+=d*x1; g2+=d*x2;
    });
    return {loss:loss/data.length,g1:g1/data.length,g2:g2/data.length};
  }
  function initTrain(start,data){
    let w1=start.w1,w2=start.w2;
    const history=[{w1,w2}];
    const steps=600,alpha=1;
    for(let k=0;k<steps;k++){
      const s=initLossGrad(w1,w2,data);
      w1-=alpha*s.g1; w2-=alpha*s.g2;
      if((k+1)%5===0) history.push({w1,w2});
    }
    return {w1,w2,loss:initLossGrad(w1,w2,data).loss,history};
  }
