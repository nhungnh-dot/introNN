  function studentExpressionToLatex(raw){

  function checkStudent(){
    // Always replace the previous student curve with the expression
    // currently shown in the math editor.
    const raw = (formulaInput.innerText || formulaInput.textContent || '').trim();

    studentFn = null;
    render();

    try{
      updateMathPreview();
      const newStudentFn = parseFormula(raw);
      studentFn = newStudentFn;

      let maxErr=0, mse=0, n=0;
      for(let i=0;i<=120;i++){
        const t=6*i/120;
        const e=studentFn(t)-exact(t);
        maxErr=Math.max(maxErr,Math.abs(e));
        mse += e*e; n++;
      }
      mse/=n;

      if(maxErr < 0.03){
        showStudentAnswer(
          '<b>Correct!</b> Your formula satisfies the ODE and the initial condition.'
        );
      }else{
        showStudentAnswer(
          '<b>Your solution is not correct yet.</b> This formula has been plotted. Check the initial condition and exponential decay rate, then revise it and try again.'
        );
      }

      render();
    }catch(err){
      studentFn=null;
      feedback.textContent = err.message;
      render();
    }
  }

  function updateTrainingReadout(){
    if(trainingStepCountEl) trainingStepCountEl.textContent = String(trainingSteps);
    if(currentLossEl) currentLossEl.textContent = observedLoss(modelA).toExponential(3);
  }

  function gradientAt(a){
    const h = 1e-4;
    return (observedLoss(a+h)-observedLoss(a-h))/(2*h);
  }

  function takeTrainingSteps(count){
    for(let k=0;k<count;k++){
      const grad = gradientAt(modelA);
      modelA -= internalLearningRate*grad;
      trainingSteps++;
    }
    modelTrained = trainingSteps > 0;
    updateTrainingReadout();

    status.innerHTML =
      '<b>Training updated.</b> The loss is now ' +
      observedLoss(modelA).toExponential(2) +
      '. The black dots are the observations, the green dashed curve is the true solution, ' +
      'the blue curve is the Neural ODE fit, and the orange dashed curve is the future prediction.';
    render();
  }

  function resetTraining(){
    modelA = -0.12;
    trainingSteps = 0;
    modelTrained = false;
    updateTrainingReadout();
    status.innerHTML =
      'Training reset. The plot now shows only the observed data points. Start with one training step to reveal the Neural ODE trajectory.';
    render();
  }

  render();
})();

  // Change these values and run the code again.
  const observationEnd = 3.0;
  const learningRate = 0.05;

  const hiddenSize = 3;

    .filter(t => t <= observationEnd + 1e-12);

  // One-hidden-layer neural network f_theta(t,z).
  // theta stores: weights from t, weights from z,
  // hidden biases, output weights, and one output bias.
  const nParams = 4*hiddenSize + 1;
  const theta = Array.from({length:nParams}, (_,i) =>
    0.18*Math.sin((i+1)*1.7)
  );

  function fTheta(t,z,p){
    let out = p[4*hiddenSize];
    for(let j=0;j<hiddenSize;j++){
      const a = p[j]*t
              + p[hiddenSize+j]*z
              + p[2*hiddenSize+j];
      const h = Math.tanh(a);
      out += p[3*hiddenSize+j]*h;
    }
    return out;
  }

  // Euler ODE solver.
  function solve(p, tMax=6){
    let t=0, z=2;
    const trajectory=[{t,z}];
    while(t < tMax-1e-12){
      const dt=Math.min(timeStep,tMax-t);
      z = z + dt*fTheta(t,z,p);
      t = t + dt;
      trajectory.push({t,z});
    }
    return trajectory;
  }
  function valueAt(trajectory,t){
    let best=trajectory[0];
    for(const point of trajectory){
      if(Math.abs(point.t-t) < Math.abs(best.t-t)) best=point;
    }
    return best.z;
  }

  function loss(p){
    const trajectory=solve(p,observationEnd);
    let sum=0;
    for(const t of observedTimes){
      const error=valueAt(trajectory,t)-exact(t);
      sum += error*error;
    }
    return sum/observedTimes.length;
  }

  // Gradient descent. To keep this browser example short,
  // derivatives are approximated with centered differences.
  const eps=1e-4;
  for(let step=0;step<trainSteps;step++){
