<script>
(function(){
  const forwardBtn = document.getElementById('odeForwardBtn');
  const backwardBtn = document.getElementById('odeBackwardBtn');
  const resetBtn = document.getElementById('odeResetFlowBtn');
  const msg = document.getElementById('odeFlowMessage');
  const blocks = [...document.querySelectorAll('.ode-flow-block')];
  const fBlocks = [...document.querySelectorAll('.ode-forward-block')];
  const bBlocks = [...document.querySelectorAll('.ode-backward-block')];
  const fArrows = [...document.querySelectorAll('.ode-forward-arrow')];
  const bArrows = [...document.querySelectorAll('.ode-backward-arrow')];
  const loop = document.getElementById('odeFeedbackPath');
  const nn = document.getElementById('odeBlockNN');

  if(!forwardBtn || !backwardBtn || !resetBtn) return;

  function clear(){
    blocks.forEach(el => el.classList.remove('ode-flow-active','ode-flow-dim'));
    fArrows.concat(bArrows).forEach(el => el.classList.remove('ode-arrow-dim'));
    if(loop) loop.classList.remove('ode-loop-dim');
  }

  forwardBtn.addEventListener('click', () => {
    clear();
    bBlocks.forEach(el => el.classList.add('ode-flow-dim'));
    bArrows.forEach(el => el.classList.add('ode-arrow-dim'));
    if(loop) loop.classList.add('ode-loop-dim');
    fBlocks.forEach(el => el.classList.add('ode-flow-active'));
    msg.innerHTML =
      '<b>Forward pass:</b> starting from the observed state, the neural network predicts the rate of change, ' +
      'the ODE solver builds a trajectory, and the trajectory is compared with the data to produce the loss.';
  });

  backwardBtn.addEventListener('click', () => {
    clear();
    fBlocks.forEach(el => {
      if(el.id !== 'odeBlockLoss' && el.id !== 'odeBlockNN') el.classList.add('ode-flow-dim');
    });
    fArrows.forEach(el => el.classList.add('ode-arrow-dim'));
    bBlocks.forEach(el => el.classList.add('ode-flow-active'));
    if(nn) nn.classList.add('ode-flow-active');
    msg.innerHTML =
      '<b>Learning update:</b> backpropagation uses the loss to compute gradients. ' +
      'Gradient descent then changes the neural-network weights and biases, which is why the arrow loops back to the neural network.';
  });

  resetBtn.addEventListener('click', () => {
    clear();
    msg.innerHTML =
      'The top row shows one complete training cycle. After gradient descent updates the parameters, ' +
      'the model repeats the forward pass with the improved neural network.';
  });
})();
</script>
